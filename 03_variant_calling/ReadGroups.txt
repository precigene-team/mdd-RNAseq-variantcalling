#!/bin/bash
#SBATCH --job-name=addRG
#SBATCH --output=logs/addRG_%j.out
#SBATCH --error=logs/addRG_%j.err
#SBATCH --cpus-per-task=4
#SBATCH --mem=32G
#SBATCH --time=48:00:00

# Go to your project directory
cd /mnt/22T/kienvu

# Activate environment
conda activate mdd

mkdir -p variant/readgroups

for bam in star/aligned_bam/*_Aligned.sortedByCoord.out.bam; do
    sample=$(basename "$bam" _Aligned.sortedByCoord.out.bam)

    gatk --java-options "-Xmx28G" AddOrReplaceReadGroups \
        -I "$bam" \
        -O "variant/readgroups/${sample}.rg.bam" \
        -RGID "$sample" \
        -RGLB lib1 \
        -RGPL ILLUMINA \
        -RGPU unit1 \
        -RGSM "$sample"
done
